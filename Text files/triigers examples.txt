| Trigger Type  | When It Executes             |
| ------------- | ---------------------------- |
| BEFORE INSERT | Before a new row is inserted |
| AFTER INSERT  | After a new row is inserted  |
| BEFORE UPDATE | Before a row is updated      |
| AFTER UPDATE  | After a row is updated       |
| BEFORE DELETE | Before a row is deleted      |
| AFTER DELETE  | After a row is deleted       |


Before Insert

DELIMITER //

CREATE TRIGGER trg_uppercase_name
BEFORE INSERT
ON Employee
FOR EACH ROW
BEGIN
    SET NEW.EmpName = UPPER(NEW.EmpName);
END //

DELIMITER ;




------------------------------------------------
After insert 

CREATE TABLE Employee_Audit (
    AuditID INT AUTO_INCREMENT PRIMARY KEY,
    Message VARCHAR(200)
);


CREATE TRIGGER trg_after_insert_employee
AFTER INSERT ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO Employee_Audit(Message)
    VALUES (CONCAT('New Employee Added: ', NEW.EmpName));
END;

Before Delete

DELIMITER //

CREATE TRIGGER trg_prevent_delete
BEFORE DELETE
ON Employee
FOR EACH ROW
BEGIN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT = 'Deletion Not Allowed';
END //

DELIMITER ;
--------------------------------------------------

CREATE TABLE Employee_Backup
AS
SELECT * FROM Employee WHERE 1 = 0;

CREATE TRIGGER trg_after_delete_employee
AFTER DELETE ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO Employee_Backup
    VALUES (
        OLD.EmpID,
        OLD.EmpName,
        OLD.Salary
    );
END;

1. BEFORE UPDATE Trigger

Prevent salary reduction.

DELIMITER //

CREATE TRIGGER trg_salary_check
BEFORE UPDATE
ON Employee
FOR EACH ROW
BEGIN
    IF  NEW.Salary < OLD.Salary THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Salary Cannot Be Reduced';
    END IF;
END //

DELIMITER ;

---------------------------------------------------------------------


After Update

CREATE TABLE Salary_Log (
    LogID INT AUTO_INCREMENT PRIMARY KEY,
    EmpID INT,
    OldSalary DECIMAL(10,2),
    NewSalary DECIMAL(10,2),
    UpdatedOn DATETIME
);



CREATE TRIGGER trg_after_update_salary
AFTER UPDATE ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO Salary_Log
    VALUES (
        NULL,
        NEW.EmpID,
        OLD.Salary,
        NEW.Salary,
        NOW()
    );
END;
